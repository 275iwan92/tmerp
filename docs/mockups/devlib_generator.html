class DeliveryMethodLookup extends MasterDataLookup
{
    public function slug(): string  { return 'md-delivery-method'; }
    public function title(): string { return 'Delivery Method'; }

    protected function query(): Builder
    {
        return RefDeliveryMethod::query()->orderByDesc('activate')->orderBy('method_code');
    }

    public function columns(): array
    {
        return [
            ['key' => 'method_code', 'label' => 'Code', 'required' => true, 'unique' => true,
             'lockSaved' => true, 'case' => 'uppercase'],
            ['key' => 'method_name', 'label' => 'Delivery Method', 'required' => true],
            ['key' => 'default_supplier_id', 'label' => 'Default Forwarder', 'type' => 'lookup',
             'lookup' => 'supplier:EKSPEDISI'],
            ['key' => 'lead_days', 'label' => 'Lead Time (days)', 'type' => 'decimal', 'decimals' => 1],
            ['key' => 'activate', 'label' => 'Active', 'type' => 'check'],
        ];
    }
    // rules(): lengths come from the table columns (varchar 10, varchar 60)
}
